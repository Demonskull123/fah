-- ============================================================
-- Core (unchanged logic from prior build)

local function getPing()
    local ok, val = pcall(function()
        return Network.ServerStatsItem["Data Ping"]:GetValue()
    end)
    if ok and type(val) == "number" then return val / 1000 end
    return 0.08
end

local function getBall()
    local folder = Workspace:FindFirstChild("Balls")
    if not folder then return nil end
    for _, v in ipairs(folder:GetChildren()) do
        if v:IsA("BasePart") then return v end
    end
    return nil
end

local function getCharacter()
    local char = LP.Character
    if not char then return nil end
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hrp or not hrp:IsA("BasePart") then return nil end
    return char, hrp
end

local function isAimedAtMe(ball, ballPos, ballVel, myPos)
    if not cfg.RequireTargeting then return true end
    local toMe = (myPos - ballPos)
    local dist = toMe.Magnitude
    if dist < 0.01 then return true end
    local dot = toMe.Unit:Dot(ballVel.Unit)
    local angle = math.deg(math.acos(math.clamp(dot, -1, 1)))
    return angle <= cfg.TargettingConeDeg
end

local function computeLead(ballPos, ballVel, myPos, ping)
    local toMe = myPos - ballPos
    local dist = toMe.Magnitude
    local speed = ballVel.Magnitude
    local closing = ballVel:Dot(toMe.Unit)
    if closing <= 0 then return nil end
    local ttHit = dist / closing
    local lead = cfg.ReactionTime
        + dist * cfg.DistanceWeight
        + speed * cfg.SpeedWeight
        + ping * cfg.PingWeight
    lead = math.clamp(lead, cfg.MinLeadTime, cfg.MaxLeadTime)
    return lead, ttHit, dist, speed
end

local function fireParry()
    local char = LP.Character
    if not char then return end
    local ok = pcall(function()
        local ReplicatedStorage = game:GetService("ReplicatedStorage")
        local Remotes = ReplicatedStorage:FindFirstChild("Remotes")
        if Remotes then
            local parry = Remotes:FindFirstChild("Parry")
            if parry and parry:IsA("RemoteEvent") then parry:FireServer() end
            local block = Remotes:FindFirstChild("Block")
            if block and block:IsA("RemoteEvent") then block:FireServer() end
        end
    end)
    if not ok then
        pcall(function()
            local VIM = game:GetService("VirtualInputManager")
            VIM:SendMouseButtonEvent(0, 0, 0, true, game, 1)
            VIM:SendMouseButtonEvent(0, 0, 0, false, game, 1)
        end)
    end
end

local running = false
local thread = nil
local lastParry = 0

local function loop()
    while getgenv().BB_AutoParry and not _G.BB_AutoParry_Kill do
        local char, hrp = getCharacter()
        if char and hrp then
            local ball = getBall()
            if ball then
                local ballPos = ball.Position
                local ballVel = ball.AssemblyLinearVelocity
                local myPos = hrp.Position
                local ping = getPing()
                local lead, ttHit, dist, speed = computeLead(ballPos, ballVel, myPos, ping)
                if lead and dist <= cfg.MaxParryDistance
                   and dist >= cfg.MinParryDistance
                   and isAimedAtMe(ball, ballPos, ballVel, myPos) then
                    local now = os.clock()
                    if now - lastParry >= cfg.Cooldown and ttHit <= lead then
                        fireParry()
                        lastParry = now
                    end
                end
            end
        end
        task.wait(cfg.PollRate)
    end
    running = false
end

local function start()
    if running then return end
    running = true
    _G.BB_AutoParry_Kill = false
    thread = task.spawn(loop)
end

local function stop()
    _G.BB_AutoParry_Kill = true
    if thread then pcall(task.cancel, thread); thread = nil end
    running = false
end

    knob.BackgroundColor3 = Color3.fromRGB(230, 230, 240)

    kCorner.CornerRadius = UDim.new(1, 0)

    local function setFromX(x)
        local rel = math.clamp((x - track.AbsolutePosition.X) / track.AbsoluteSize.X, 0, 1)
        local raw = minV + rel * (maxV - minV)
        local snapped = math.floor((raw - minV) / step + 0.5) * step + minV
        snapped = math.clamp(snapped, minV, maxV)
        cfg[key] = snapped
        fill.Size = UDim2.new(rel, 0, 1, 0)
        knob.Position = UDim2.new(rel, -6, 0.5, -6)
        lbl.Text = labelText .. "  —  " .. string.format(fmt, snapped)
    end

    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            draggingSlider = true
            setFromX(input.Position.X)
        end
    end)

    UIS.InputChanged:Connect(function(input)
        if draggingSlider and (input.UserInputType == Enum.UserInputType.MouseMovement
        or input.UserInputType == Enum.UserInputType.Touch) then
            setFromX(input.Position.X)
        end
    end)

    UIS.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
        or input.UserInputType == Enum.UserInputType.Touch then
            draggingSlider = false
        end
    end)
end

    pcall(function() getgenv().BB_AutoParry_Watcher:Disconnect() end)
end

    pcall(function() getgenv().BB_AutoParry_HotkeyConn:Disconnect() end)

local last = getgenv().BB_AutoParry

getgenv().BB_AutoParry_Watcher = RunService.Heartbeat:Connect(function()
    local cur = getgenv().BB_AutoParry
    if cur == last then return end
    last = cur
    if cur then start() else stop() end
end)

getgenv().BB_AutoParry_HotkeyConn = UIS.InputBegan:Connect(function(input, gpe)
    if gpe then return end
    if input.KeyCode == getgenv().BB_AutoParryHotkey then
        getgenv().BB_AutoParry = not getgenv().BB_AutoParry
    end
end)

if getgenv().BB_AutoParry then start() end

getgenv().BB_AutoParry_Unload = function()
    getgenv().BB_AutoParry = false
    stop()
    if getgenv().BB_AutoParry_Watcher then
        pcall(function() getgenv().BB_AutoParry_Watcher:Disconnect() end)
        getgenv().BB_AutoParry_Watcher = nil
    end
    if getgenv().BB_AutoParry_HotkeyConn then
        pcall(function() getgenv().BB_AutoParry_HotkeyConn:Disconnect() end)
        getgenv().BB_AutoParry_HotkeyConn = nil
    end
    if getgenv().BB_AutoParry_UI then
        pcall(function() getgenv().BB_AutoParry_UI:Destroy() end)
        getgenv().BB_AutoParry_UI = nil
    end
end

return true
